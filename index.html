# Extracting differential capacity curves
def extract_dq_dv_features(voltage, capacity):
    # Numerical derivative of Q (capacity) with respect to V (voltage)
    dq_dv = np.gradient(capacity) / np.gradient(voltage)
    # Extract peak height, peak location, and curve variance
    peak_val = np.max(dq_dv)
    peak_loc = voltage[np.argmax(dq_dv)]
    return {
        'peak_dq_dv': peak_val,
        'voltage_at_peak': peak_loc,
        'variance': np.var(dq_dv)
    }
